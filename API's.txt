1.Authentication APis 

    1.1Login (POST) :  "http://localhost:3001/api/v1/auth/login"  

        1.1.1 Request Data:{ 

            email:  (String Email) , 

            password: (String) 

        } 

        

        1.1.2 Response   

            1.1.2.1 Corrects Credentials: Status – 200 – ok  

                { 
                    Success: true, 
                    Message: "login Successfully" 
                } 

            1.1.2.2 Wrong Credentials: status- 404 

                {
                    "message": "user not found"
                } 

    1.2 Register (Post) : "http://localhost:3001/api/v1/auth/register" 

        1.2.1 Request Data:{ 

                username: (string),

                age: (int),

                password: (String),

                confirmPassword: (String),

                email:  (String Email) , 

                role: (String)

            } 

        1.2.2 Response   

            1.2.2.1 Corrects Credentials: Status – 200 – ok  

                { 
                    Success: true, 
                    Message: "user registered Successfully" 
                } 

            1.2.2.2 Duplicate Entry: status- 409

                {
                    "message": 'user with this email already exist'
                } 

            1.2.2.3 Large Data Value: status- 400

                {
                    "message": 'Input value too long' 
                }
                
            1.2.2.4 missig field: status- 400

                {
                    "message": 'Required field is missing' 
                } 

                
    1.3 logout (get) : "http://localhost:3001/api/v1/auth/logout" 

        1.3.1 Request Data:{ 

                no data Required, Data has been taken from the cookies

            } 

        1.3.2 Response   

            1.3.2.1 no cookies in headers: status – 401  

                {  
                    Message:'User is already logged out'
                } 

            1.3.2.2 token inside cookie: status- 200 - ok

                {
                    "message": "succesfully logged out"
                } 

            1.3.2.3 if no token inside the cookie: status - 400

                {
                    "message": "error loggingout"
                }

    1.4 emailOtpRequest (post) : "http://localhost:3001/api/v1/auth/email-otp-request"

        1.4.1 Request Data:{

            1.4.2.1 user Email not found in database: status – 400

                {
                      message:"sign-in with google"

            1.4.2.3 otp sent to the user email: status- 200 - ok

                     message: "otp sended"
